\newcommand{\bigcomp@}[2]{%
  \begingroup
  \m@th
  \sbox\z@{$#1\sum$}%
  \setlength{\unitlength}{0.9\dimexpr\ht\z@+\dp\z@}%
  \vcenter{%
    \hbox{%
      \begin{picture}(1,1)
        \bigcomp@linethickness{#1}
        \put(0.5,0.5){\circle{1}}
      \end{picture}%
    }%
  }%
  \endgroup
}
\newcommand{\bigcomp@linethickness}[1]{%
  \ifx#1\displaystyle
    \linethickness{0.09em}%
  \else\ifx#1\textstyle
    \linethickness{0.075em}%
  \else\ifx#1\scriptstyle
    \linethickness{0.06em}%
  \else
    \linethickness{0.05em}%
  \fi\fi\fi
}
